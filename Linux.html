// 微服务架构核心实现
class MicroService {
  constructor(name, port) {
    this.name = name;
    this.port = port;
    this.healthCheck = () => {
      return { 
        status: 'healthy', 
        uptime: process.uptime() 
      };
    };
  }

  start() {
    const server = http.createServer(this.handleRequest);
    server.listen(this.port, () => {
      console.log(`Service ${this.name} running on port ${this.port}`);
    });
  }

  handleRequest(req, res) {
    res.endJSON({ 
      message: 'OK', 
      service: this.name 
    });
  }
}

webgl-renderer.js 
// WebGL 渲染器核心类
class WebGLRenderer {
  constructor(canvas) {
    this.gl = canvas.getContext('webgl2');
    this.initShaders();
    this.createBuffers();
  }

  initShaders() {
    const vsSource = `
      attribute vec3 aPosition;
      void main() {
        gl_Position = vec4(aPosition, 1.0);
      }
    `;
    
    const fsSource = `
      precision mediump float;
      void main() {
        gl_FragColor = vec4(0.0, 1.0, 0.6, 1.0);
      }
    `;
    
    this.program = this.createProgram(vsSource, fsSource);
  }

  render(vertices) {
    this.gl.clearColor(0.0, 0.0, 0.0, 1.0);
    this.gl.clear(gl.COLOR_BUFFER_BIT);
    this.gl.useProgram(this.program);
    this.gl.bufferData(gl.ARRAY_BUFFER, vertices, gl.STATIC_DRAW);
    this.gl.drawArrays(gl.TRIANGLES, 0, 3);
  }
}

// 异步任务调度器
class TaskScheduler {
  constructor(concurrency = 3) {
    this.concurrency = concurrency;
    this.queue = [];
    this.running = 0;
  }

  async schedule(task) {
    return new Promise((resolve, reject) => {
      this.queue.push({ task, resolve, reject });
      this.process();
    });
  }

  async process() {
    while (this.running < 3 && this.queue.length > 0) {
      const { task, resolve, reject } = this.queue.shift();
      this.running++;
      
      try {
        const result = await task();
        resolve(result);
      } catch (error) {
        reject(error);
      } finally {
        this.running--;
      }
    }
  }
}

// 使用示例
const scheduler = new TaskScheduler(3);
const results = await Promise.all(
  tasks.map(task => scheduler.schedule(task))
);
